// ═══ 1. DEPARTMENT ═══
model Department {
  id        String   @id @default(uuid())
  code      String   @unique
  name      String
  description String? 
  active    Boolean  @default(true)
  createdAt DateTime @default(now()) @map("created_at")
  updatedAt DateTime @updatedAt @map("updated_at")

  users   User[]
  tickets Ticket[]

  @@map("departments")
}

// ═══ 2. USER ═══
model User {
  id           String    @id @default(uuid())
  username     String    @unique
  passwordHash String    @map("password_hash")
  fullName     String    @map("full_name")
  email        String?   @unique
  phone        String?
  role         String
  position     String?
  departmentId String?   @map("department_id")
  department   Department? @relation(fields: [departmentId], references: [id])
  active       Boolean   @default(true)
  lastLoginAt  DateTime? @map("last_login_at")
  createdAt    DateTime  @default(now()) @map("created_at")
  updatedAt    DateTime  @updatedAt @map("updated_at")
  deletedAt    DateTime? @map("deleted_at")

  createdTickets  Ticket[]      @relation("TicketCreator")
  assignedTickets Ticket[]      @relation("TicketAssignee")
  replies         TicketReply[]

  @@index([role])
  @@index([departmentId])
  @@map("users")
}

// ═══ 3. TICKET ═══
model Ticket {
  id          String  @id @default(uuid())
  code        String  @unique
  title       String
  description String?
  category    String
  priority    String  @default("NORMAL")
  status      String  @default("NEW")

  creatorId       String     @map("creator_id")
  creator         User       @relation("TicketCreator", fields: [creatorId], references: [id])
  creatorName     String     @map("creator_name")
  creatorPhone    String?    @map("creator_phone")
  creatorPosition String?    @map("creator_position")

  departmentId   String     @map("department_id")
  department     Department @relation(fields: [departmentId], references: [id])
  departmentName String     @map("department_name")

  assigneeId   String? @map("assignee_id")
  assignee     User?   @relation("TicketAssignee", fields: [assigneeId], references: [id])
  assigneeName String? @map("assignee_name")

  dispatcherId String?   @map("dispatcher_id")
  dispatchedAt DateTime? @map("dispatched_at")

  scheduledAt DateTime? @map("scheduled_at")
  completedAt DateTime? @map("completed_at")
  processNote String?   @map("process_note")

  rating   Int?
  feedback String?
  ratedAt  DateTime? @map("rated_at")

  supplies    TicketSupply[]
  attachments TicketAttachment[]
  replies     TicketReply[]

  createdAt DateTime @default(now()) @map("created_at")
  updatedAt DateTime @updatedAt @map("updated_at")

  @@index([status])
  @@index([creatorId])
  @@index([assigneeId])
  @@index([departmentId])
  @@map("tickets")
}

// ═══ 4. TICKET_SUPPLY ═══
model TicketSupply {
  id        String   @id @default(uuid())
  ticketId  String   @map("ticket_id")
  ticket    Ticket   @relation(fields: [ticketId], references: [id], onDelete: Cascade)
  name      String
  quantity  Int      @default(1)
  unit      String   @default("Cái")
  note      String?
  createdAt DateTime @default(now()) @map("created_at")

  @@index([ticketId])
  @@map("ticket_supplies")
}

// ═══ 5. TICKET_ATTACHMENT ═══
model TicketAttachment {
  id         String   @id @default(uuid())
  ticketId   String   @map("ticket_id")
  ticket     Ticket   @relation(fields: [ticketId], references: [id], onDelete: Cascade)
  fileName   String   @map("file_name")
  filePath   String   @map("file_path")
  fileSize   Int      @map("file_size")
  fileType   String   @map("file_type")
  stage      String   @default("BEFORE")
  uploadedBy String   @map("uploaded_by")
  uploadedAt DateTime @default(now()) @map("uploaded_at")

  @@index([ticketId])
  @@map("ticket_attachments")
}

// ═══ 6. TICKET_REPLY ═══
model TicketReply {
  id        String   @id @default(uuid())
  ticketId  String   @map("ticket_id")
  ticket    Ticket   @relation(fields: [ticketId], references: [id], onDelete: Cascade)
  userId    String   @map("user_id")
  user      User     @relation(fields: [userId], references: [id])
  userName  String   @map("user_name")
  userRole  String   @map("user_role")
  content   String
  createdAt DateTime @default(now()) @map("created_at")

  @@index([ticketId])
  @@map("ticket_replies")
}

// ═══ 7. AUDIT_LOG ═══
model AuditLog {
  id         String   @id @default(uuid())
  userId     String?  @map("user_id")
  userName   String?  @map("user_name")
  userRole   String?  @map("user_role")
  action     String
  entityType String?  @map("entity_type")
  entityId   String?  @map("entity_id")
  oldValue   Json?    @map("old_value")
  newValue   Json?    @map("new_value")
  ipAddress  String?  @map("ip_address")
  level      String   @default("INFO")
  createdAt  DateTime @default(now()) @map("created_at")

  @@index([userId])
  @@index([action])
  @@index([createdAt])
  @@map("audit_logs")
}

import multer from 'multer';
import path from 'path';
import fs from 'fs';
import { fileURLToPath } from 'url';

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
const UPLOAD_DIR = path.resolve(__dirname, '../../uploads');

if (!fs.existsSync(UPLOAD_DIR)) {
  fs.mkdirSync(UPLOAD_DIR, { recursive: true });
}

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    const ticketId = req.params.id || 'temp';
    const dir = path.join(UPLOAD_DIR, ticketId);
    if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
    cb(null, dir);
  },
  filename: (req, file, cb) => {
    const id = `att-${Date.now()}-${Math.floor(Math.random() * 1000)}`;
    const ext = path.extname(file.originalname);
    const baseName = path.basename(file.originalname, ext).replace(/[^a-zA-Z0-9]/g, '_');
    cb(null, `${id}_${baseName}${ext}`);
  },
});

const ALLOWED = [
  'image/jpeg', 'image/png', 'image/gif', 'image/webp',
  'application/pdf',
  'application/msword',
  'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
  'application/vnd.ms-excel',
  'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
];

export const upload = multer({
  storage,
  limits: { fileSize: 20 * 1024 * 1024 },
  fileFilter: (req, file, cb) => {
    if (ALLOWED.includes(file.mimetype)) cb(null, true);
    else cb(new Error(`Loại file không được phép: ${file.mimetype}`));
  },
});

export { UPLOAD_DIR };

import jwt from 'jsonwebtoken';
import prisma from '../config/prisma.js';

export async function authenticate(req, res, next) {
  let token = null;

  const authHeader = req.headers.authorization;
  if (authHeader?.startsWith('Bearer ')) token = authHeader.slice(7).trim();
  if (!token && req.query.token) token = String(req.query.token);

  if (!token) {
    return res.status(401).json({ success: false, message: 'Chưa đăng nhập' });
  }

  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    const user = await prisma.user.findUnique({
      where: { id: decoded.id },
      include: { department: { select: { id: true, code: true, name: true } } },
    });

    if (!user || user.deletedAt || !user.active) {
      return res.status(401).json({ success: false, message: 'Tài khoản không hợp lệ' });
    }

    req.user = {
      id: user.id,
      username: user.username,
      fullName: user.fullName,
      role: user.role,
      departmentId: user.departmentId,
      department: user.department,
      position: user.position,
      phone: user.phone,
    };
    next();
  } catch (err) {
    return res.status(401).json({ success: false, message: 'Token không hợp lệ hoặc hết hạn' });
  }
}

export function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({ success: false, message: 'Chưa đăng nhập' });
    }
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({
        success: false,
        message: `Chỉ ${roles.join(', ')} mới có quyền truy cập`,
      });
    }
    next();
  };
}

  }, 25000);

  req.on('close', () => {
    clearInterval(heartbeat);
    sseService.removeClient(userId, res);
    res.end();
  });
});

    success: true,

import prisma from '../config/prisma.js';
import fs from 'fs';

export const attachmentsService = {
  async upload(ticketId, file, { stage = 'BEFORE' }, currentUser) {
    if (!file) throw { status: 400, message: 'Chưa chọn file' };

    const ticket = await prisma.ticket.findUnique({ where: { id: ticketId } });
    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };

    if (currentUser.role === 'CUSTOMER' && ticket.creatorId !== currentUser.id) {
      throw { status: 403, message: 'Không có quyền' };
    }

    const attachment = await prisma.ticketAttachment.create({
      data: {
        ticketId,
        fileName: file.originalname,
        filePath: file.path,
        fileSize: file.size,
        fileType: file.mimetype,
        stage,
        uploadedBy: currentUser.id,
      },
    });

    return attachment;
  },

  async list(ticketId) {
    return prisma.ticketAttachment.findMany({
      where: { ticketId },
      orderBy: { uploadedAt: 'asc' },
    });
  },

  async getForDownload(attachmentId) {
    const att = await prisma.ticketAttachment.findUnique({ where: { id: attachmentId } });
    if (!att) throw { status: 404, message: 'Không tìm thấy file' };
    if (!fs.existsSync(att.filePath)) {
      throw { status: 404, message: 'File không tồn tại trên server' };
    }
    return att;
  },

  async remove(attachmentId, currentUser) {
    const att = await prisma.ticketAttachment.findUnique({ where: { id: attachmentId } });
    if (!att) throw { status: 404, message: 'Không tìm thấy file' };

    if (att.uploadedBy !== currentUser.id && currentUser.role !== 'ADMIN') {
      throw { status: 403, message: 'Không có quyền xóa file này' };
    }

    try { fs.unlinkSync(att.filePath); } catch (e) { /* ignore */ }
    await prisma.ticketAttachment.delete({ where: { id: attachmentId } });

    return { success: true, message: 'Đã xóa file' };
  },
};

        skip,
        take: limit,

import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import prisma from '../config/prisma.js';
import { auditLogsService } from './auditLogs.service.js';

export const authService = {
  async login({ username, password, ip }) {
    if (!username || !password) {
      throw { status: 400, message: 'Vui lòng nhập tên đăng nhập và mật khẩu' };
    }

    const user = await prisma.user.findUnique({
      where: { username: username.trim().toLowerCase() },
      include: { department: { select: { id: true, code: true, name: true } } },
    });

    if (!user || user.deletedAt) {
      throw { status: 401, message: 'Tên đăng nhập hoặc mật khẩu không chính xác' };
    }
    if (!user.active) {
      throw { status: 403, message: 'Tài khoản đã bị khóa' };
    }

    const match = await bcrypt.compare(password, user.passwordHash);
    if (!match) {
      await auditLogsService.create({
        userId: user.id,
        userName: user.fullName,
        userRole: user.role,
        action: 'AUTH_LOGIN_FAILED',
        entityType: 'USER',
        entityId: user.id,
        ipAddress: ip,
        level: 'SECURITY',
      });
      throw { status: 401, message: 'Tên đăng nhập hoặc mật khẩu không chính xác' };
    }

    await prisma.user.update({
      where: { id: user.id },
      data: { lastLoginAt: new Date() },
    });

    const token = jwt.sign(
      { id: user.id, username: user.username, role: user.role },
      process.env.JWT_SECRET,
      { expiresIn: process.env.JWT_EXPIRES_IN || '7d' }
    );

    await auditLogsService.create({
      userId: user.id,
      userName: user.fullName,
      userRole: user.role,
      action: 'AUTH_LOGIN',
      entityType: 'USER',
      entityId: user.id,
      ipAddress: ip,
      level: 'INFO',
    });

    const { passwordHash, ...safe } = user;
    return { success: true, user: safe, token };
  },

  async me(userId) {
    const user = await prisma.user.findUnique({
      where: { id: userId },
      include: { department: { select: { id: true, code: true, name: true } } },
    });
    if (!user) throw { status: 404, message: 'Không tìm thấy user' };
    const { passwordHash, ...safe } = user;
    return safe;
  },

  async changePassword(userId, oldPassword, newPassword) {
    if (!oldPassword || !newPassword) {
      throw { status: 400, message: 'Vui lòng nhập đầy đủ mật khẩu cũ và mới' };
    }
    if (newPassword.length < 6) {
      throw { status: 400, message: 'Mật khẩu mới phải có ít nhất 6 ký tự' };
    }

    const user = await prisma.user.findUnique({ where: { id: userId } });
    if (!user) throw { status: 404, message: 'Không tìm thấy user' };

    const match = await bcrypt.compare(oldPassword, user.passwordHash);
    if (!match) throw { status: 401, message: 'Mật khẩu hiện tại không chính xác' };

    const newHash = await bcrypt.hash(newPassword, 10);
    await prisma.user.update({
      where: { id: userId },
      data: { passwordHash: newHash },
    });

    await auditLogsService.create({
      userId,
      userName: user.fullName,
      userRole: user.role,
      action: 'CHANGE_PASSWORD',
      entityType: 'USER',
      entityId: userId,
      level: 'SECURITY',
    });

    return { success: true, message: 'Đổi mật khẩu thành công' };
  },
};

import prisma from '../config/prisma.js';
import { auditLogsService } from './auditLogs.service.js';

export const departmentsService = {
  async list(filters = {}) {
    const where = {};
    if (filters.active !== undefined) where.active = filters.active === 'true';
    if (filters.search) {
      where.OR = [
        { code: { contains: filters.search, mode: 'insensitive' } },
        { name: { contains: filters.search, mode: 'insensitive' } },
      ];
    }

    const departments = await prisma.department.findMany({
      where,
      orderBy: { code: 'asc' },
      include: { _count: { select: { users: true, tickets: true } } },
    });

    return { departments };
  },

  async getById(id) {
    const dept = await prisma.department.findUnique({
      where: { id },
      include: { users: { select: { id: true, username: true, fullName: true, role: true } } },
    });
    if (!dept) throw { status: 404, message: 'Không tìm thấy phòng ban' };
    return dept;
  },

  async create(data, currentUser) {
    const { code, name, description } = data; // <-- Thêm description vào đây
    if (!code || !name) throw { status: 400, message: 'Thiếu mã hoặc tên phòng' };

    const existing = await prisma.department.findUnique({ where: { code: code.trim().toUpperCase() } });
    if (existing) throw { status: 400, message: `Mã phòng "${code}" đã tồn tại` };

    const dept = await prisma.department.create({
      data: { code: code.trim().toUpperCase(), name: name.trim(), description: description?.trim() || null },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'CREATE_DEPARTMENT',
      entityType: 'DEPARTMENT',
      entityId: dept.id,
      newValue: { code, name, description },
      level: 'INFO',
    });

    return dept;
  },

  async update(id, data, currentUser) {
    const existing = await prisma.department.findUnique({ where: { id } });
    if (!existing) throw { status: 404, message: 'Không tìm thấy phòng ban' };

    const updateData = {};
    if (data.name !== undefined) updateData.name = data.name.trim();
    if (data.description !== undefined) updateData.description = data.description?.trim() || null;   // ✅ thêm
    if (data.active !== undefined) updateData.active = data.active;

    const dept = await prisma.department.update({ where: { id }, data: updateData });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'UPDATE_DEPARTMENT',
      entityType: 'DEPARTMENT',
      entityId: id,
      newValue: updateData,
      level: 'INFO',
    });

    return dept;
  },

  async remove(id, currentUser) {
    const dept = await prisma.department.findUnique({
      where: { id },
      include: { _count: { select: { users: true, tickets: true } } },
    });
    if (!dept) throw { status: 404, message: 'Không tìm thấy phòng ban' };

    if (dept._count.users > 0) {
      throw { status: 400, message: `Còn ${dept._count.users} user trong phòng` };
    }
    if (dept._count.tickets > 0) {
      throw { status: 400, message: `Còn ${dept._count.tickets} phiếu thuộc phòng` };
    }

    await prisma.department.update({ where: { id }, data: { active: false } });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'DELETE_DEPARTMENT',
      entityType: 'DEPARTMENT',
      entityId: id,
      oldValue: { code: dept.code, name: dept.name },
      level: 'SECURITY',
    });

    return { success: true, message: 'Đã xóa phòng ban' };
  },
};

import { sseService } from './sse.service.js';

export function notifyRealtime(userIds, eventData) {
  if (!Array.isArray(userIds) || userIds.length === 0) return;

  setImmediate(() => {
    try {
      sseService.sendToUsers(userIds, 'notification', {
        id: `notif-${Date.now()}`,
        ...eventData,
        isRead: false,
        createdAt: new Date().toISOString(),
      });
    } catch (err) {
      console.error('❌ Lỗi SSE:', err.message);
    }
  });
}

class SSEService {
  constructor() {
    this.clients = new Map();
  }

  addClient(userId, res) {
    if (!this.clients.has(userId)) this.clients.set(userId, new Set());
    this.clients.get(userId).add(res);
    console.log(`📡 [SSE] + user ${userId}. Total: ${this.getTotalConnections()}`);
  }

  removeClient(userId, res) {
    const set = this.clients.get(userId);
    if (set) {
      set.delete(res);
      if (set.size === 0) this.clients.delete(userId);
    }
  }

  sendToUser(userId, eventType, data) {
    const set = this.clients.get(userId);
    if (!set || set.size === 0) return 0;

    const payload = JSON.stringify({ type: eventType, data, timestamp: new Date().toISOString() });
    let sent = 0;
    set.forEach((res) => {
      try {
        res.write(`event: ${eventType}\n`);
        res.write(`data: ${payload}\n\n`);
        sent++;
      } catch (err) {
        set.delete(res);
      }
    });
    return sent;
  }

  sendToUsers(userIds, eventType, data) {
    let total = 0;
    userIds.forEach((id) => { total += this.sendToUser(id, eventType, data); });
    return total;
  }

  broadcast(eventType, data) {
    let total = 0;
    this.clients.forEach((set, userId) => {
      total += this.sendToUser(userId, eventType, data);
    });
    return total;
  }

  getTotalConnections() {
    let count = 0;
    this.clients.forEach((set) => { count += set.size; });
    return count;
  }

  getOnlineUserCount() { return this.clients.size; }
}

export const sseService = new SSEService();

import prisma from '../config/prisma.js';
import { auditLogsService } from './auditLogs.service.js';
import { notifyRealtime } from './notification.helper.js';

async function generateCode() {
  const year = new Date().getFullYear();
  const prefix = `PYC-${year}-`;
  const last = await prisma.ticket.findFirst({
    where: { code: { startsWith: prefix } },
    orderBy: { code: 'desc' },
    select: { code: true },
  });
  const nextNum = last ? parseInt(last.code.split('-').pop()) + 1 : 1;
  return `${prefix}${String(nextNum).padStart(3, '0')}`;
}

export const ticketsService = {
  async list(currentUser, filters = {}) {
    const page = parseInt(filters.page) || 1;
    const limit = parseInt(filters.limit) || 20;
    const skip = (page - 1) * limit;

    const where = {};

    if (currentUser.role === 'CUSTOMER') {
      where.creatorId = currentUser.id;
    } else if (currentUser.role === 'TECHNICAL') {
      if (filters.onlyMine === 'true') where.assigneeId = currentUser.id;
    }

    if (filters.status) where.status = filters.status;
    if (filters.priority) where.priority = filters.priority;
    if (filters.category) where.category = filters.category;
    if (filters.assigneeId) where.assigneeId = filters.assigneeId;
    if (filters.departmentId) where.departmentId = filters.departmentId;

    if (filters.search) {
      where.OR = [
        { code: { contains: filters.search, mode: 'insensitive' } },
        { title: { contains: filters.search, mode: 'insensitive' } },
        { creatorName: { contains: filters.search, mode: 'insensitive' } },
      ];
    }

    const [tickets, total] = await Promise.all([
      prisma.ticket.findMany({
        where,
        skip,
        take: limit,
        orderBy: [{ priority: 'desc' }, { createdAt: 'desc' }],
        include: {
          supplies: true,
          _count: { select: { attachments: true, replies: true } },
        },
      }),
      prisma.ticket.count({ where }),
    ]);

    return { tickets, pagination: { page, limit, total, totalPages: Math.ceil(total / limit) } };
  },

  async getById(id, currentUser) {
    const ticket = await prisma.ticket.findUnique({
      where: { id },
      include: {
        supplies: true,
        attachments: { orderBy: { uploadedAt: 'asc' } },
        replies: { orderBy: { createdAt: 'asc' } },
      },
    });

    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };

    if (currentUser.role === 'CUSTOMER' && ticket.creatorId !== currentUser.id) {
      throw { status: 403, message: 'Không có quyền xem phiếu này' };
    }

    return ticket;
  },

  async create(data, currentUser) {
    if (currentUser.role !== 'CUSTOMER') {
      throw { status: 403, message: 'Chỉ đơn vị (CUSTOMER) mới được tạo phiếu' };
    }

    const {
      title, description, category, priority,
      creatorName, creatorPhone, creatorPosition,
      supplies = [],
    } = data;

    if (!title || !category) {
      throw { status: 400, message: 'Thiếu tiêu đề hoặc loại sự cố' };
    }
    if (!currentUser.departmentId) {
      throw { status: 400, message: 'Tài khoản chưa được gán đơn vị' };
    }

    const dept = await prisma.department.findUnique({ where: { id: currentUser.departmentId } });
    const code = await generateCode();

    const ticket = await prisma.ticket.create({
      data: {
        code,
        title: title.trim(),
        description: description?.trim() || null,
        category,
        priority: priority || 'NORMAL',
        status: 'NEW',
        creatorId: currentUser.id,
        creatorName: creatorName || currentUser.fullName,
        creatorPhone: creatorPhone || currentUser.phone,
        creatorPosition: creatorPosition || currentUser.position,
        departmentId: currentUser.departmentId,
        departmentName: dept?.name || '',
        supplies: supplies.length > 0 ? {
          create: supplies.map((s) => ({
            name: s.name,
            quantity: s.quantity || 1,
            unit: s.unit || 'Cái',
            note: s.note || null,
          })),
        } : undefined,
      },
      include: { supplies: true },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'CREATE_TICKET',
      entityType: 'TICKET',
      entityId: ticket.id,
      newValue: { code, title, priority },
      level: 'INFO',
    });

    const admins = await prisma.user.findMany({
      where: { role: 'ADMIN', active: true, deletedAt: null },
      select: { id: true },
    });
    notifyRealtime(admins.map((a) => a.id), {
      type: 'TICKET_NEW',
      title: '📋 Phiếu mới cần điều phối',
      content: `${ticket.code} — ${ticket.title}`,
      ticketId: ticket.id,
    });

    return ticket;
  },

  async assign(id, { assigneeId }, currentUser) {
    if (currentUser.role !== 'ADMIN') {
      throw { status: 403, message: 'Chỉ ADMIN được điều phối phiếu' };
    }

    const ticket = await prisma.ticket.findUnique({ where: { id } });
    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };
    if (ticket.status === 'DONE' || ticket.status === 'CANCELLED') {
      throw { status: 400, message: 'Phiếu đã kết thúc' };
    }

    const assignee = await prisma.user.findUnique({ where: { id: assigneeId } });
    if (!assignee || assignee.role !== 'TECHNICAL' || !assignee.active) {
      throw { status: 400, message: 'Người được gán phải là TECHNICAL đang hoạt động' };
    }

    const updated = await prisma.ticket.update({
      where: { id },
      data: {
        assigneeId,
        assigneeName: assignee.fullName,
        dispatcherId: currentUser.id,
        dispatchedAt: new Date(),
        status: 'ASSIGNED',
      },
      include: { supplies: true },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'ASSIGN_TICKET',
      entityType: 'TICKET',
      entityId: id,
      newValue: { assigneeId, assigneeName: assignee.fullName },
      level: 'INFO',
    });

    notifyRealtime([assigneeId], {
      type: 'TICKET_ASSIGNED',
      title: '🎯 Bạn được giao phiếu mới',
      content: `${ticket.code} — ${ticket.title}`,
      ticketId: id,
    });
    notifyRealtime([ticket.creatorId], {
      type: 'TICKET_STATUS',
      title: '✅ Phiếu đã được điều phối',
      content: `${ticket.code} đã được giao cho ${assignee.fullName}`,
      ticketId: id,
    });

    return updated;
  },

  async process(id, data, currentUser) {
    if (currentUser.role !== 'TECHNICAL') {
      throw { status: 403, message: 'Chỉ TECHNICAL mới cập nhật xử lý' };
    }

    const ticket = await prisma.ticket.findUnique({ where: { id } });
    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };
    if (ticket.assigneeId !== currentUser.id) {
      throw { status: 403, message: 'Bạn không được gán phiếu này' };
    }
    if (ticket.status === 'DONE' || ticket.status === 'CANCELLED') {
      throw { status: 400, message: 'Phiếu đã kết thúc' };
    }

    const { status, scheduledAt, processNote, supplies } = data;
    const updateData = {};

    if (status && ['ASSIGNED', 'PROCESSING', 'WAITING_SUPPLY', 'DONE'].includes(status)) {
      updateData.status = status;
    }
    if (scheduledAt !== undefined) {
      updateData.scheduledAt = scheduledAt ? new Date(scheduledAt) : null;
    }
    if (processNote !== undefined) updateData.processNote = processNote;
    if (status === 'DONE' && !ticket.completedAt) {
      updateData.completedAt = new Date();
    }

    if (Array.isArray(supplies)) {
      await prisma.ticketSupply.deleteMany({ where: { ticketId: id } });
      if (supplies.length > 0) {
        await prisma.ticketSupply.createMany({
          data: supplies.map((s) => ({
            ticketId: id,
            name: s.name,
            quantity: s.quantity || 1,
            unit: s.unit || 'Cái',
            note: s.note || null,
          })),
        });
      }
    }

    const updated = await prisma.ticket.update({
      where: { id },
      data: updateData,
      include: { supplies: true },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'PROCESS_TICKET',
      entityType: 'TICKET',
      entityId: id,
      newValue: updateData,
      level: 'INFO',
    });

    if (status && status !== ticket.status) {
      const statusLabel = {
        PROCESSING: 'đang xử lý',
        WAITING_SUPPLY: 'đang chờ vật tư',
        DONE: 'đã hoàn thành',
        ASSIGNED: 'đã được điều phối',
      }[status] || status;

      notifyRealtime([ticket.creatorId], {
        type: 'TICKET_STATUS',
        title: `📢 Phiếu ${ticket.code} ${statusLabel}`,
        content: status === 'DONE'
          ? 'Vui lòng vào đánh giá chất lượng dịch vụ!'
          : 'Kỹ thuật viên đã cập nhật trạng thái',
        ticketId: id,
      });
    }

    return updated;
  },

  async evaluate(id, { rating, feedback }, currentUser) {
    if (currentUser.role !== 'CUSTOMER') {
      throw { status: 403, message: 'Chỉ đơn vị tạo phiếu mới được đánh giá' };
    }

    const ticket = await prisma.ticket.findUnique({ where: { id } });
    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };
    if (ticket.creatorId !== currentUser.id) {
      throw { status: 403, message: 'Bạn không phải chủ phiếu này' };
    }
    if (ticket.status !== 'DONE') {
      throw { status: 400, message: 'Chỉ đánh giá được phiếu đã hoàn thành' };
    }
    if (ticket.rating) {
      throw { status: 400, message: 'Phiếu đã được đánh giá' };
    }
    if (!rating || rating < 1 || rating > 5) {
      throw { status: 400, message: 'Đánh giá phải từ 1 đến 5 sao' };
    }

    const updated = await prisma.ticket.update({
      where: { id },
      data: { rating, feedback: feedback?.trim() || null, ratedAt: new Date() },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'EVALUATE_TICKET',
      entityType: 'TICKET',
      entityId: id,
      newValue: { rating, feedback },
      level: 'INFO',
    });

    return updated;
  },

  async addReply(id, { content }, currentUser) {
    if (!content?.trim()) throw { status: 400, message: 'Nội dung trả lời không được trống' };

    const ticket = await prisma.ticket.findUnique({ where: { id } });
    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };

    if (currentUser.role === 'CUSTOMER' && ticket.creatorId !== currentUser.id) {
      throw { status: 403, message: 'Không có quyền' };
    }

    const reply = await prisma.ticketReply.create({
      data: {
        ticketId: id,
        userId: currentUser.id,
        userName: currentUser.fullName,
        userRole: currentUser.role,
        content: content.trim(),
      },
    });

    return reply;
  },

  async stats(currentUser) {
    const where = {};
    if (currentUser.role === 'CUSTOMER') where.creatorId = currentUser.id;
    else if (currentUser.role === 'TECHNICAL') where.assigneeId = currentUser.id;

    const all = await prisma.ticket.findMany({
      where,
      select: { status: true, rating: true, priority: true, createdAt: true, completedAt: true },
    });

    const total = all.length;
    const byStatus = {
      NEW: 0, ASSIGNED: 0, PROCESSING: 0, WAITING_SUPPLY: 0, DONE: 0, CANCELLED: 0,
    };
    all.forEach((t) => { byStatus[t.status] = (byStatus[t.status] || 0) + 1; });

    const rated = all.filter((t) => t.rating);
    const avgRating = rated.length > 0
      ? rated.reduce((s, t) => s + t.rating, 0) / rated.length
      : 0;

    return {
      total,
      byStatus,
      avgRating: Math.round(avgRating * 10) / 10,
      ratedCount: rated.length,
    };
  },

  async update(id, data, currentUser) {
    if (currentUser.role !== 'ADMIN') {
      throw { status: 403, message: 'Chỉ ADMIN mới được sửa phiếu' };
    }

    const existing = await prisma.ticket.findUnique({ where: { id } });
    if (!existing) throw { status: 404, message: 'Không tìm thấy phiếu' };

    const {
      title, description, category, priority, status,
      creatorName, creatorPhone, creatorPosition,
      assigneeId, scheduledAt, processNote,
    } = data;

    const updateData = {};

    if (title !== undefined) updateData.title = title.trim();
    if (description !== undefined) updateData.description = description?.trim() || null;
    if (category !== undefined) updateData.category = category;
    if (priority !== undefined) updateData.priority = priority;
    if (status !== undefined) updateData.status = status;
    if (creatorName !== undefined) updateData.creatorName = creatorName;
    if (creatorPhone !== undefined) updateData.creatorPhone = creatorPhone;
    if (creatorPosition !== undefined) updateData.creatorPosition = creatorPosition;
    if (processNote !== undefined) updateData.processNote = processNote;

    // Nếu đổi assignee
    if (assigneeId !== undefined) {
      if (assigneeId === null) {
        updateData.assigneeId = null;
        updateData.assigneeName = null;
      } else {
        const assignee = await prisma.user.findUnique({ where: { id: assigneeId } });
        if (!assignee || assignee.role !== 'TECHNICAL') {
          throw { status: 400, message: 'Người được gán phải là TECHNICAL' };
        }
        updateData.assigneeId = assigneeId;
        updateData.assigneeName = assignee.fullName;
      }
    }

    if (scheduledAt !== undefined) {
      updateData.scheduledAt = scheduledAt ? new Date(scheduledAt) : null;
    }

    // Nếu status → DONE thì set completedAt
    if (status === 'DONE' && !existing.completedAt) {
      updateData.completedAt = new Date();
    }

    const updated = await prisma.ticket.update({
      where: { id },
      data: updateData,
      include: { supplies: true },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'UPDATE_TICKET',
      entityType: 'TICKET',
      entityId: id,
      oldValue: { code: existing.code, title: existing.title, status: existing.status },
      newValue: updateData,
      level: 'INFO',
    });

    return updated;
  },

  // ✅ THÊM MỚI: ADMIN xóa ticket (soft delete ở đây là xóa thật, cascade xóa supplies/attachments/replies)
  async remove(id, currentUser) {
    if (currentUser.role !== 'ADMIN') {
      throw { status: 403, message: 'Chỉ ADMIN mới được xóa phiếu' };
    }

    const ticket = await prisma.ticket.findUnique({ where: { id } });
    if (!ticket) throw { status: 404, message: 'Không tìm thấy phiếu' };

    // Xóa file vật lý của attachments trước
    try {
      const attachments = await prisma.ticketAttachment.findMany({ where: { ticketId: id } });
      const fs = await import('fs');
      for (const att of attachments) {
        try { fs.unlinkSync(att.filePath); } catch (e) { /* ignore */ }
      }
    } catch (e) {
      console.error('Lỗi xóa file attachments:', e.message);
    }

    // Xóa ticket (cascade sẽ tự xóa supplies, attachments, replies)
    await prisma.ticket.delete({ where: { id } });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'DELETE_TICKET',
      entityType: 'TICKET',
      entityId: id,
      oldValue: { code: ticket.code, title: ticket.title, status: ticket.status },
      level: 'SECURITY',
    });

    return { success: true, message: `Đã xóa phiếu [${ticket.code}]` };
  },

};

import bcrypt from 'bcrypt';
import prisma from '../config/prisma.js';
import { auditLogsService } from './auditLogs.service.js';

const DEFAULT_PASSWORD = 'vks@2026';

export const usersService = {
  async list(filters = {}) {
    const page = parseInt(filters.page) || 1;
    const limit = parseInt(filters.limit) || 20;
    const skip = (page - 1) * limit;

    const where = { deletedAt: null };
    if (filters.role) where.role = filters.role;
    if (filters.departmentId) where.departmentId = filters.departmentId;
    if (filters.active !== undefined) where.active = filters.active === 'true';
    if (filters.search) {
      where.OR = [
        { username: { contains: filters.search, mode: 'insensitive' } },
        { fullName: { contains: filters.search, mode: 'insensitive' } },
        { email: { contains: filters.search, mode: 'insensitive' } },
      ];
    }

    const [users, total] = await Promise.all([
      prisma.user.findMany({
        where,
        skip,
        take: limit,
        orderBy: { createdAt: 'desc' },
        select: {
          id: true, username: true, fullName: true, email: true, phone: true,
          role: true, position: true, active: true, lastLoginAt: true,
          createdAt: true, departmentId: true,
          department: { select: { id: true, code: true, name: true } },
        },
      }),
      prisma.user.count({ where }),
    ]);

    return { users, pagination: { page, limit, total, totalPages: Math.ceil(total / limit) } };
  },

  async getById(id) {
    const user = await prisma.user.findUnique({
      where: { id },
      include: { department: { select: { id: true, code: true, name: true } } },
    });
    if (!user || user.deletedAt) throw { status: 404, message: 'Không tìm thấy user' };
    const { passwordHash, ...safe } = user;
    return safe;
  },

  async create(data, currentUser) {
    const { username, password, fullName, email, phone, role, position, departmentId } = data;

    if (!username || !fullName || !role) {
      throw { status: 400, message: 'Thiếu username, họ tên hoặc vai trò' };
    }
    if (!['ADMIN', 'CUSTOMER', 'TECHNICAL'].includes(role)) {
      throw { status: 400, message: 'Vai trò không hợp lệ' };
    }

    const existing = await prisma.user.findUnique({
      where: { username: username.trim().toLowerCase() },
    });
    if (existing) throw { status: 400, message: 'Username đã tồn tại' };

    if (email) {
      const emailExists = await prisma.user.findUnique({ where: { email } });
      if (emailExists) throw { status: 400, message: 'Email đã tồn tại' };
    }

    const passwordHash = await bcrypt.hash(password || DEFAULT_PASSWORD, 10);

    const user = await prisma.user.create({
      data: {
        username: username.trim().toLowerCase(),
        passwordHash,
        fullName: fullName.trim(),
        email: email || null,
        phone: phone || null,
        role,
        position: position || null,
        departmentId: departmentId || null,
      },
      include: { department: { select: { id: true, code: true, name: true } } },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'CREATE_USER',
      entityType: 'USER',
      entityId: user.id,
      newValue: { username, fullName, role },
      level: 'SECURITY',
    });

    const { passwordHash: _, ...safe } = user;
    return safe;
  },

  async update(id, data, currentUser) {
    const existing = await prisma.user.findUnique({ where: { id } });
    if (!existing || existing.deletedAt) {
      throw { status: 404, message: 'Không tìm thấy user' };
    }

    const updateData = {};
    const allowed = ['fullName', 'email', 'phone', 'role', 'position', 'departmentId', 'active'];
    allowed.forEach((f) => {
      if (data[f] !== undefined) updateData[f] = data[f];
    });

    if (data.password && data.password.trim().length >= 6) {
      updateData.passwordHash = await bcrypt.hash(data.password.trim(), 10);
    }

    const updated = await prisma.user.update({
      where: { id },
      data: updateData,
      include: { department: { select: { id: true, code: true, name: true } } },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'UPDATE_USER',
      entityType: 'USER',
      entityId: id,
      oldValue: { fullName: existing.fullName },
      newValue: updateData,
      level: 'INFO',
    });

    const { passwordHash, ...safe } = updated;
    return safe;
  },

  async remove(id, currentUser) {
    const user = await prisma.user.findUnique({ where: { id } });
    if (!user || user.deletedAt) throw { status: 404, message: 'Không tìm thấy user' };
    if (user.id === currentUser.id) throw { status: 400, message: 'Không thể xóa chính mình' };
    if (user.username === 'admin') throw { status: 400, message: 'Không thể xóa admin gốc' };

    const stamp = Date.now();
    await prisma.user.update({
      where: { id },
      data: {
        active: false,
        deletedAt: new Date(),
        username: `${user.username}__deleted_${stamp}`,
        email: user.email ? `${user.email}__deleted_${stamp}` : null,
      },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: 'DELETE_USER',
      entityType: 'USER',
      entityId: id,
      oldValue: { username: user.username, fullName: user.fullName },
      level: 'SECURITY',
    });

    return { success: true, message: `Đã xóa tài khoản "${user.fullName}"` };
  },

  async toggleActive(id, currentUser) {
    const user = await prisma.user.findUnique({ where: { id } });
    if (!user) throw { status: 404, message: 'Không tìm thấy user' };
    if (user.id === currentUser.id) throw { status: 400, message: 'Không thể khóa chính mình' };
    if (user.username === 'admin') throw { status: 400, message: 'Không thể khóa admin gốc' };

    const updated = await prisma.user.update({
      where: { id },
      data: { active: !user.active },
    });

    await auditLogsService.create({
      userId: currentUser.id,
      userName: currentUser.fullName,
      userRole: currentUser.role,
      action: updated.active ? 'UNLOCK_USER' : 'LOCK_USER',
      entityType: 'USER',
      entityId: id,
      level: 'SECURITY',
    });

    return {
      success: true,
      message: updated.active ? 'Đã mở khóa' : 'Đã khóa',
      active: updated.active,
    };
  },
};
