import React, { useState, useEffect } from 'react';
import { AuthProvider, useAuth } from './context/AuthContext';
import { ToastProvider } from './components/common/Toast';
import { Login } from './pages/Login';
import { AdminDashboard } from './pages/AdminDashboard';
import { CustomerDashboard } from './pages/CustomerDashboard';
import { TechnicalDashboard } from './pages/TechnicalDashboard';

function AppContent() {
  const { isAuthenticated } = useAuth();
  const [currentPath, setCurrentPath] = useState<string>(() => {
    return window.location.pathname || '/';
  });

  // Keep path synced with browser URL
  useEffect(() => {
    const handlePopState = () => {
      setCurrentPath(window.location.pathname);
    };
    window.addEventListener('popstate', handlePopState);
    return () => window.removeEventListener('popstate', handlePopState);
  }, []);

  const navigate = (path: string) => {
    window.history.pushState({}, '', path);
    setCurrentPath(path);
  };

  // Route protection (only check authentication, NO role restriction)
  useEffect(() => {
    if (!isAuthenticated) {
      if (currentPath !== '/login') {
        navigate('/login');
      }
    } else {
      if (currentPath === '/login') {
        navigate('/');
      }
    }
  }, [isAuthenticated, currentPath]);

  // Render view according to path
  if (!isAuthenticated || currentPath === '/login') {
    return <Login onNavigate={navigate} />;
  }

  // Without role barriers, any user can view any interface:
  if (currentPath === '/customer') {
    return <CustomerDashboard />;
  }

  if (currentPath === '/technical') {
    return <TechnicalDashboard />;
  }

  // Default is the complete unified workspace
  return <AdminDashboard />;
}

export default function App() {
  return (
    <ToastProvider>
      <AuthProvider>
        <AppContent />
      </AuthProvider>
    </ToastProvider>
  );
}

  tickets: Ticket[];

  onGoToDispatch: () => void;
  onGoToAllTickets: (statusFilter?: string) => void; // ✅ SỬA: nhận statusFilter
}

  onViewTicket,
  onGoToDispatch,
  onGoToAllTickets,

  const newCount = tickets.filter((t) => t.status === 'NEW').length;
  const processingCount = tickets.filter(
    (t) => t.status === 'PROCESSING' || t.status === 'ASSIGNED' || t.status === 'WAITING_SUPPLY'
  ).length;
  const doneCount = tickets.filter((t) => t.status === 'DONE').length;

  const statusChartData = useMemo(() => {
    const counts: Record<string, number> = {
      NEW: 0,
      ASSIGNED: 0,
      PROCESSING: 0,
      WAITING_SUPPLY: 0,
      DONE: 0,
      CANCELLED: 0,
    };

    tickets.forEach((t) => {
      if (counts[t.status] !== undefined) {
        counts[t.status]++;
      }
    });

    const colors: Record<string, string> = {
      NEW: '#94a3b8',
      ASSIGNED: '#a855f7',
      PROCESSING: '#3b82f6',
      WAITING_SUPPLY: '#f59e0b',
      DONE: '#10b981',
      CANCELLED: '#f43f5e',
    };

    return Object.entries(counts).map(([statusKey, count]) => ({
      name: STATUS_CONFIG[statusKey as keyof typeof STATUS_CONFIG]?.label || statusKey,
      value: count,
      color: colors[statusKey] || '#ccc',
    }));
  }, [tickets]);

  const last7DaysData = useMemo(() => {
    const days: Array<{ day: string; created: number; processing: number; done: number }> = [];
    const now = new Date();

    for (let i = 6; i >= 0; i--) {
      const d = new Date(now);
      d.setDate(d.getDate() - i);
      const dayStr = d.toLocaleDateString('vi-VN', {
        weekday: 'short',
        day: 'numeric',
        month: 'numeric',
      });
      const datePrefix = d.toISOString().slice(0, 10);

      const createdCount = tickets.filter((t) => t.createdAt.startsWith(datePrefix)).length;

      const processingCountOnDay = tickets.filter(
        (t) =>
          (t.status === 'PROCESSING' || t.status === 'ASSIGNED' || t.status === 'WAITING_SUPPLY') &&
          t.updatedAt &&
          t.updatedAt.startsWith(datePrefix)
      ).length;

      const doneCountOnDay = tickets.filter(
        (t) => t.status === 'DONE' && t.completedAt && t.completedAt.startsWith(datePrefix)
      ).length;

      days.push({
        day: dayStr,
        created: createdCount,
        processing: processingCountOnDay,
        done: doneCountOnDay,
      });
    }

    return days;
  }, [tickets]);

  const recentTickets = useMemo(() => {
    return [...tickets]
      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
      .slice(0, 5);
  }, [tickets]);

  return (
    <div className="space-y-6">
      {/* 4 KPI Cards */}
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
        {/* Total — Click → ALL */}
        <div
          onClick={() => onGoToAllTickets('ALL')}
          className="bg-white rounded-2xl p-5 border border-neutral-200/90 shadow-2xs relative overflow-hidden cursor-pointer hover:border-neutral-500 hover:shadow-md transition-all group"
        >
          <div className="flex items-center justify-between">
            <span className="text-xs font-bold text-neutral-500 uppercase tracking-wider">
              Tổng số phiếu
            </span>
            <div className="w-9 h-9 rounded-xl bg-neutral-100 text-neutral-700 flex items-center justify-center group-hover:scale-110 transition-transform">
              <FileText className="w-5 h-5" />
            </div>
          </div>
          <div className="mt-3 flex items-baseline gap-2">
            <span className="text-3xl font-black text-neutral-900 tracking-tight">
              {totalCount}
            </span>
            <span className="text-xs text-neutral-400">phiếu yêu cầu</span>
          </div>
          <div className="mt-2 text-[11px] text-neutral-500 flex items-center gap-1">
            <TrendingUp className="w-3.5 h-3.5 text-blue-600" />
            Toàn bộ các đơn vị trực thuộc
          </div>
        </div>

        {/* Chờ điều phối — Click → Dispatch */}
        <div
          onClick={onGoToDispatch}
          className="bg-white rounded-2xl p-5 border border-red-200/90 shadow-2xs relative overflow-hidden cursor-pointer hover:border-[#B71C1C] hover:shadow-md transition-all group"
        >
          <div className="flex items-center justify-between">
            <span className="text-xs font-bold text-[#B71C1C] uppercase tracking-wider">
              Chờ điều phối
            </span>
            <div className="w-9 h-9 rounded-xl bg-red-50 text-[#B71C1C] flex items-center justify-center group-hover:scale-110 transition-transform">
              <Clock className="w-5 h-5" />
            </div>
          </div>
          <div className="mt-3 flex items-baseline gap-2">
            <span className="text-3xl font-black text-[#B71C1C] tracking-tight">
              {newCount}
            </span>
            <span className="text-xs text-red-600 font-semibold">cần xử lý ngay</span>
          </div>
          <div className="mt-2 text-[11px] text-red-600 flex items-center gap-1 font-medium">
            <span>Bấm để phân công KTV</span>
            <ArrowUpRight className="w-3.5 h-3.5" />
          </div>
        </div>

        {/* Đang xử lý — Click → All Tickets filter PROCESSING */}
        <div
          onClick={() => onGoToAllTickets('PROCESSING')}
          className="bg-white rounded-2xl p-5 border border-blue-200/90 shadow-2xs relative overflow-hidden cursor-pointer hover:border-blue-600 hover:shadow-md transition-all group"
        >
          <div className="flex items-center justify-between">
            <span className="text-xs font-bold text-blue-600 uppercase tracking-wider">

              <Wrench className="w-5 h-5" />

            <span className="text-3xl font-black text-blue-900 tracking-tight">
              {processingCount}

            <span className="text-xs text-neutral-400">phiếu đang thực hiện</span>

            <ArrowUpRight className="w-3.5 h-3.5" />

        {/* Hoàn thành — Click → All Tickets filter DONE */}
        <div
          onClick={() => onGoToAllTickets('DONE')}
          className="bg-white rounded-2xl p-5 border border-emerald-200/90 shadow-2xs relative overflow-hidden cursor-pointer hover:border-emerald-600 hover:shadow-md transition-all group"

import React, { useState } from 'react';
import { AuditLog, AuditLogLevel } from '../../types';
import { ShieldAlert, Search, Filter } from 'lucide-react';

interface AuditLogTableProps {
  logs: AuditLog[];
}

export const AuditLogTable: React.FC<AuditLogTableProps> = ({ logs }) => {
  const [searchTerm, setSearchTerm] = useState('');
  const [levelFilter, setLevelFilter] = useState<string>('ALL');

  const filteredLogs = logs.filter((log) => {
    const matchSearch =
      (log.userName || '').toLowerCase().includes(searchTerm.toLowerCase()) ||
      log.action.toLowerCase().includes(searchTerm.toLowerCase()) ||
      (log.entityType || '').toLowerCase().includes(searchTerm.toLowerCase()) ||
      (log.ipAddress || '').toLowerCase().includes(searchTerm.toLowerCase());
    const matchLevel = levelFilter === 'ALL' || log.level === levelFilter;

    return matchSearch && matchLevel;
  });

  const getLevelBadge = (level: AuditLogLevel) => {
    switch (level) {
      case 'INFO':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-full text-[11px] font-bold bg-blue-100 text-blue-800 border border-blue-200">
            INFO
          </span>
        );
      case 'WARNING':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-full text-[11px] font-bold bg-amber-100 text-amber-800 border border-amber-200">
            WARNING
          </span>
        );
      case 'SECURITY':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-full text-[11px] font-bold bg-purple-100 text-purple-800 border border-purple-200">
            SECURITY
          </span>
        );
      case 'DANGER':
        return (
          <span className="inline-flex items-center px-2 py-0.5 rounded-full text-[11px] font-bold bg-rose-100 text-rose-800 border border-rose-200">
            DANGER
          </span>
        );
    }
  };

  const formatDate = (dateStr: string) => {
    try {
      return new Date(dateStr).toLocaleString('vi-VN', {
        day: '2-digit',
        month: '2-digit',
        year: 'numeric',
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
      });
    } catch {
      return dateStr;
    }
  };

  return (
    <div className="space-y-4">
      {/* Header & Filter */}
      <div className="bg-white p-5 rounded-2xl border border-neutral-200/90 shadow-xs flex flex-col sm:flex-row sm:items-center justify-between gap-3">
        <div>
          <h2 className="text-lg font-bold text-neutral-900 flex items-center gap-2">
            <ShieldAlert className="w-5 h-5 text-[#B71C1C]" />
            Nhật ký kiểm toán hệ thống (Audit Logs)
          </h2>
          <p className="text-xs text-neutral-500 mt-0.5">
            Lưu vết tự động tất cả các thao tác nghiệp vụ, điều phối, thay đổi trạng thái và an ninh.
          </p>
        </div>

        <div className="flex items-center gap-2">
          {/* Search */}
          <div className="relative">
            <Search className="w-3.5 h-3.5 text-neutral-400 absolute left-3 top-1/2 -translate-y-1/2" />
            <input
              type="text"
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
              placeholder="Tìm hành động, người dùng..."
              className="pl-8 pr-3 py-1.5 text-xs bg-neutral-50 border border-neutral-200 rounded-xl focus:border-[#B71C1C] focus:outline-none w-52"
            />
          </div>

          {/* Level filter */}
          <select
            value={levelFilter}
            onChange={(e) => setLevelFilter(e.target.value)}
            className="text-xs bg-neutral-50 border border-neutral-200 rounded-xl px-2.5 py-1.5 text-neutral-700 font-medium focus:outline-none cursor-pointer"
          >
            <option value="ALL">Tất cả mức độ</option>
            <option value="INFO">INFO (Xanh)</option>
            <option value="WARNING">WARNING (Vàng)</option>
            <option value="SECURITY">SECURITY (Tím)</option>
            <option value="DANGER">DANGER (Đỏ)</option>
          </select>
        </div>
      </div>

      {/* Logs Table */}
      <div className="bg-white rounded-2xl border border-neutral-200/90 shadow-xs overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse text-xs">
            <thead>
              <tr className="bg-neutral-50/80 border-b border-neutral-200 text-neutral-500 font-semibold uppercase tracking-wider">
                <th className="py-3 px-4 w-40">Thời gian</th>
                <th className="py-3 px-4 w-48">Người thực hiện</th>
                <th className="py-3 px-4">Hành động ghi nhận</th>
                <th className="py-3 px-4 w-44">Đối tượng (Entity)</th>
                <th className="py-3 px-4 w-32 font-mono">Địa chỉ IP</th>
                <th className="py-3 px-4 w-24 text-center">Mức độ</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-neutral-100 font-normal">
              {filteredLogs.length === 0 ? (
                <tr>
                  <td colSpan={6} className="py-8 text-center text-neutral-400">
                    Không tìm thấy bản ghi nhật ký phù hợp.
                  </td>
                </tr>
              ) : (
                filteredLogs.map((log) => (
                  <tr key={log.id} className="hover:bg-neutral-50/80 transition-colors">
                    <td>{formatDate(log.createdAt)}</td>
                    <td>{log.userName}</td>
                    <td>{log.action}</td>
                    <td>{log.entityType} {log.entityId && `(${log.entityId})`}</td>
                    <td>{log.ipAddress}</td>
                    <td className="py-3 px-4 text-center whitespace-nowrap">
                      {getLevelBadge(log.level)}
                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};

import React, { useState } from 'react';
import { Department } from '../../types';
import { apiClient } from '../../services/apiClient';
import { useToast } from '../common/Toast';
import { Modal } from '../common/Modal';
import { Building, Plus, Edit2, Trash2 } from 'lucide-react';

interface DeptManagerProps {
  departments: Department[];
  onRefresh: () => void;
}

export const DeptManager: React.FC<DeptManagerProps> = ({ departments, onRefresh }) => {
  const toast = useToast();

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editingDept, setEditingDept] = useState<Department | null>(null);
  const [code, setCode] = useState('');
  const [name, setName] = useState('');
  const [description, setDescription] = useState('');

  const openCreateModal = () => {
    setEditingDept(null);
    setCode('');
    setName('');
    setDescription('');
    setIsModalOpen(true);
  };

  const openEditModal = (dept: Department) => {
    setEditingDept(dept);
    setCode(dept.code);
    setName(dept.name);
    setDescription(dept.description || '');
    setIsModalOpen(true);
  };

  const handleSave = async (e: React.FormEvent) => {
    e.preventDefault();

    if (!code.trim() || !name.trim()) {
      toast.error('Vui lòng nhập mã và tên đơn vị!');
      return;
    }

    try {
      if (editingDept) {
        const res = await apiClient.updateDepartment(editingDept.id, {
          name: name.trim(),
          description: description.trim() || undefined,
        });
        if (res.success) {
          toast.success(`Cập nhật đơn vị [${code.toUpperCase()}] thành công!`);
        } else {
          toast.error('Lỗi cập nhật đơn vị');
          return;
        }
      } else {
        if (departments.some((d) => d.code.toUpperCase() === code.trim().toUpperCase())) {
          toast.error('Mã đơn vị này đã tồn tại!');
          return;
        }

        const res = await apiClient.createDepartment({
          code: code.trim().toUpperCase(),
          name: name.trim(),
          description: description.trim() || undefined,
        });

        if (res.success) {
          toast.success(`Thêm đơn vị [${code.toUpperCase()}] thành công!`);
        } else {
          toast.error('Lỗi thêm đơn vị');
          return;
        }
      }

      setIsModalOpen(false);
      onRefresh();
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    }
  };

  const handleDelete = async (dept: Department) => {
    if (!confirm(`Bạn có chắc muốn xóa đơn vị "${dept.name}" (${dept.code})?`)) return;

    try {
      const res = await apiClient.deleteDepartment(dept.id);
      if (res.success) {
        toast.success(`Đã xóa đơn vị [${dept.code}]`);
        onRefresh();
      } else {
        toast.error(res.message || 'Lỗi xóa đơn vị');
      }
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    }
  };

  return (
    <div className="space-y-4">
      <div className="bg-white p-5 rounded-2xl border border-neutral-200/90 flex flex-col sm:flex-row sm:items-center justify-between gap-3">
        <div>
          <h2 className="text-lg font-bold text-neutral-900 flex items-center gap-2">
            <Building className="w-5 h-5 text-[#B71C1C]" />
            Quản lý Đơn vị / Phòng ban
          </h2>
          <p className="text-xs text-neutral-500 mt-0.5">
            Danh mục các phòng ban, đơn vị trực thuộc có phát sinh yêu cầu kỹ thuật.
          </p>
        </div>

        <button
          type="button"
          onClick={openCreateModal}
          className="inline-flex items-center gap-2 px-4 py-2.5 bg-[#B71C1C] hover:bg-[#7F0E0E] text-white font-bold text-xs rounded-xl shadow-xs transition-colors cursor-pointer"
        >
          <Plus className="w-4 h-4 text-[#FFD700]" />
          <span>+ Thêm đơn vị</span>
        </button>
      </div>

      <div className="bg-white rounded-2xl border border-neutral-200/90 shadow-xs overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse text-xs">
            <thead>
              <tr className="bg-neutral-50/80 border-b border-neutral-200 text-neutral-500 font-semibold uppercase tracking-wider">
                <th className="py-3 px-4 w-32">Mã đơn vị</th>
                <th className="py-3 px-4">Tên đơn vị / Phòng ban</th>
                <th className="py-3 px-4">Mô tả chức năng</th>
                <th className="py-3 px-4 text-center w-28">Thao tác</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-neutral-100">
              {departments.map((d) => (
                <tr key={d.id} className="hover:bg-neutral-50 transition-colors">
                  <td className="py-3.5 px-4 font-mono font-bold text-[#B71C1C]">
                    <span className="bg-red-50 px-2 py-0.5 rounded border border-red-100">
                      {d.code}
                    </span>
                  </td>
                  <td className="py-3.5 px-4 font-semibold text-neutral-900">
                    {d.name}
                  </td>
                  <td className="py-3.5 px-4 text-neutral-500">
                    {d.description || '---'}
                  </td>
                  <td className="py-3.5 px-4 text-center whitespace-nowrap">
                    <div className="flex items-center justify-center gap-2">
                      <button
                        type="button"
                        onClick={() => openEditModal(d)}
                        className="p-1.5 text-neutral-600 hover:text-blue-600 hover:bg-neutral-100 rounded-lg transition-colors cursor-pointer"
                        title="Sửa"
                      >
                        <Edit2 className="w-3.5 h-3.5" />
                      </button>
                      <button
                        type="button"
                        onClick={() => handleDelete(d)}
                        className="p-1.5 text-neutral-400 hover:text-red-600 hover:bg-red-50 rounded-lg transition-colors cursor-pointer"
                        title="Xóa"
                      >
                        <Trash2 className="w-3.5 h-3.5" />
                      </button>
                    </div>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>

      <Modal
        isOpen={isModalOpen}
        onClose={() => setIsModalOpen(false)}
        title={editingDept ? 'Chỉnh sửa đơn vị' : 'Thêm đơn vị mới'}
        subtitle="Quản lý mã viết tắt và tên phòng ban trong viện"
        maxWidth="md"
      >
        <form onSubmit={handleSave} className="space-y-4">
          <div>
            <label className="block text-xs font-bold text-neutral-700 uppercase mb-1">
              Mã đơn vị (Viết tắt) <span className="text-red-500">*</span>
            </label>
            <input
              type="text"
              required
              value={code}
              onChange={(e) => setCode(e.target.value.toUpperCase())}
              disabled={!!editingDept}
              placeholder="ví dụ: P1, P2, VP, TCKT..."
              className="w-full px-3 py-2 text-xs border border-neutral-300 rounded-xl focus:border-[#B71C1C] focus:outline-none font-mono font-bold disabled:bg-neutral-100"
            />
          </div>

          <div>
            <label className="block text-xs font-bold text-neutral-700 uppercase mb-1">
              Tên đầy đủ của đơn vị <span className="text-red-500">*</span>
            </label>
            <input
              type="text"
              required
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder="ví dụ: Phòng 1 (THQCT & KSĐT Án hình sự)..."
              className="w-full px-3 py-2 text-xs border border-neutral-300 rounded-xl focus:border-[#B71C1C] focus:outline-none"
            />
          </div>

          <div>
            <label className="block text-xs font-bold text-neutral-700 uppercase mb-1">
              Mô tả chi tiết (Tùy chọn)
            </label>
            <textarea
              rows={3}
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              placeholder="Mô tả nhiệm vụ, vị trí phòng..."
              className="w-full px-3 py-2 text-xs border border-neutral-300 rounded-xl focus:border-[#B71C1C] focus:outline-none"
            />
          </div>

          <div className="flex justify-end gap-2.5 pt-3 border-t border-neutral-100">
            <button
              type="button"
              onClick={() => setIsModalOpen(false)}
              className="px-4 py-2 text-xs font-semibold text-neutral-600 hover:bg-neutral-100 rounded-xl cursor-pointer"
            >
              Hủy
            </button>
            <button
              type="submit"
              className="px-5 py-2 text-xs font-bold text-white bg-[#B71C1C] hover:bg-[#7F0E0E] rounded-xl shadow-xs cursor-pointer"
            >
              {editingDept ? 'LƯU THAY ĐỔI' : 'TẠO ĐƠN VỊ'}
            </button>
          </div>
        </form>
      </Modal>
    </div>
  );
};

import React, { useState } from 'react';
import { Ticket, User } from '../../types';
import { PriorityBadge } from '../common/StatusBadge';
import { apiClient } from '../../services/apiClient';
import { useToast } from '../common/Toast';
import { useAuth } from '../../context/AuthContext';
import { GitPullRequest, UserCheck, Building2, CheckCircle } from 'lucide-react';

interface DispatchBoardProps {
  tickets: Ticket[];
  technicians: User[];
  onDispatched?: () => void;
}

export const DispatchBoard: React.FC<DispatchBoardProps> = ({
  tickets,
  technicians,
  onDispatched,
}) => {
  const toast = useToast();
  const newTickets = tickets.filter((t) => t.status === 'NEW');

  const [selectedTicketId, setSelectedTicketId] = useState<string | null>(
    newTickets.length > 0 ? newTickets[0].id : null
  );
  const [selectedTechId, setSelectedTechId] = useState<string>(
    technicians.length > 0 ? technicians[0].id : ''
  );
  const [scheduledTime, setScheduledTime] = useState<string>('');
  const [isAssigning, setIsAssigning] = useState(false);

  const activeTicket = newTickets.find((t) => t.id === selectedTicketId)
    || (newTickets.length > 0 ? newTickets[0] : null);

  const handleAssign = async () => {
    if (!activeTicket) {
      toast.error('Vui lòng chọn phiếu yêu cầu cần điều phối!');
      return;
    }
    if (!selectedTechId) {
      toast.error('Vui lòng chọn kỹ thuật viên để phân công!');
      return;
    }

    const tech = technicians.find((t) => t.id === selectedTechId);
    if (!tech) return;

    setIsAssigning(true);
    try {
      const res = await apiClient.assignTicket(activeTicket.id, selectedTechId);
      if (res.success) {
        toast.success(`Đã phân công phiếu [${activeTicket.code}] cho ${tech.fullName}`);
        const remaining = newTickets.filter((t) => t.id !== activeTicket.id);
        setSelectedTicketId(remaining.length > 0 ? remaining[0].id : null);
        if (onDispatched) onDispatched();
      } else {
        toast.error(res.message || 'Lỗi điều phối');
      }
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    } finally {
      setIsAssigning(false);
    }
  };

  const formatDate = (dateStr?: string) => {
    if (!dateStr) return '---';
    try {
      return new Date(dateStr).toLocaleString('vi-VN', {
        day: '2-digit', month: '2-digit', year: 'numeric',
        hour: '2-digit', minute: '2-digit',
      });
    } catch { return dateStr; }
  };

  return (
    <div className="space-y-4">
      <div className="bg-white p-5 rounded-2xl border border-neutral-200/90 flex flex-col sm:flex-row sm:items-center justify-between gap-3">
        <div>
          <h2 className="text-lg font-bold text-neutral-900 flex items-center gap-2">
            <GitPullRequest className="w-5 h-5 text-[#B71C1C]" />
            Bảng điều phối kỹ thuật
          </h2>
          <p className="text-xs text-neutral-500 mt-0.5">
            Tiếp nhận phiếu mới và phân công cho kỹ thuật viên phụ trách.
          </p>
        </div>
        <div className="inline-flex items-center gap-2 px-3 py-1.5 rounded-xl bg-red-50 text-[#B71C1C] border border-red-200 text-xs font-bold">
          <span>Đang chờ phân công:</span>
          <span className="w-5 h-5 rounded-full bg-[#B71C1C] text-white flex items-center justify-center text-xs">
            {newTickets.length}
          </span>
        </div>
      </div>

      {newTickets.length === 0 ? (
        <div className="bg-white rounded-2xl p-12 text-center border border-neutral-200/90 shadow-xs">
          <div className="w-14 h-14 rounded-2xl bg-emerald-50 text-emerald-600 flex items-center justify-center mx-auto mb-3">
            <CheckCircle className="w-8 h-8" />
          </div>
          <h3 className="font-bold text-neutral-800 text-base">Không còn phiếu chờ điều phối</h3>
          <p className="text-xs text-neutral-500 mt-1">
            Tất cả phiếu đã được phân công xử lý.
          </p>
        </div>
      ) : (
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
          <div className="lg:col-span-5 space-y-3">
            <div className="text-xs font-bold text-neutral-400 uppercase tracking-wider px-1">
              Danh sách phiếu mới ({newTickets.length})
            </div>
            <div className="space-y-2.5 max-h-[620px] overflow-y-auto pr-1">
              {newTickets.map((t) => {
                const isSelected = activeTicket?.id === t.id;
                return (
                  <div
                    key={t.id}
                    onClick={() => setSelectedTicketId(t.id)}
                    className={`p-4 rounded-2xl border transition-all cursor-pointer ${
                      isSelected
                        ? 'bg-red-50/70 border-[#B71C1C] shadow-sm'
                        : 'bg-white border-neutral-200/90 hover:border-neutral-300'
                    }`}
                  >
                    <div className="flex items-center justify-between mb-2">
                      <span className="font-mono font-bold text-xs bg-neutral-100 text-neutral-800 px-2 py-0.5 rounded">
                        {t.code}
                      </span>
                      <PriorityBadge priority={t.priority} size="sm" />
                    </div>
                    <h4 className="font-bold text-sm text-neutral-900 line-clamp-1 mb-1">
                      {t.title}

                        <Building2 className="w-3.5 h-3.5 text-neutral-400 shrink-0" />
                        <span className="truncate">{t.departmentName}</span>
                      </div>
                      <div className="flex items-center justify-between">
                        <span className="text-neutral-700 font-medium">{t.creatorName}</span>
                        <span className="text-[11px] text-neutral-400">{formatDate(t.createdAt)}</span>
                      </div>

              })}
            </div>
          </div>

          <div className="lg:col-span-7">
            {activeTicket && (
              <div className="bg-white rounded-2xl p-6 border border-neutral-200/90 shadow-sm space-y-6 sticky top-20">
                <div className="flex items-center justify-between border-b border-neutral-100 pb-3">
                  <div>
                    <span className="font-mono font-bold text-xs bg-red-100 text-[#B71C1C] px-2.5 py-1 rounded-md">
                      {activeTicket.code}
                    </span>
                    <h3 className="text-base font-bold text-neutral-900 mt-2">
                      {activeTicket.title}
                    </h3>
                  </div>
                  <PriorityBadge priority={activeTicket.priority} size="md" />
                </div>

                <div className="bg-neutral-50 rounded-xl p-3.5 border border-neutral-200/70 grid grid-cols-2 gap-3 text-xs">
                  <div>
                    <span className="text-neutral-400 block">Đơn vị gửi:</span>
                    <span className="font-semibold text-neutral-800">{activeTicket.departmentName}</span>
                  </div>
                  <div>
                    <span className="text-neutral-400 block">Người gửi:</span>
                    <span className="font-semibold text-neutral-800">
                      {activeTicket.creatorName} ({activeTicket.creatorPosition})
                    </span>
                  </div>
                  <div>
                    <span className="text-neutral-400 block">SĐT:</span>
                    <span className="font-semibold text-blue-600">{activeTicket.creatorPhone}</span>
                  </div>
                  <div>
                    <span className="text-neutral-400 block">Loại sự cố:</span>
                    <span className="font-semibold text-[#B71C1C]">{activeTicket.category}</span>

import React, { useState } from 'react';

import { useAuth } from '../../context/AuthContext';

import {

} from 'lucide-react';

interface UserManagerProps {
  users: User[];
  departments: Department[];
  onRefresh: () => void;
}

export const UserManager: React.FC<UserManagerProps> = ({ users, departments, onRefresh }) => {
  const { user: currentAdmin } = useAuth();
  const toast = useToast();

  const [isModalOpen, setIsModalOpen] = useState(false);
  const [editingUser, setEditingUser] = useState<User | null>(null);

  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [fullName, setFullName] = useState('');
  const [email, setEmail] = useState('');
  const [phone, setPhone] = useState('');
  const [role, setRole] = useState<'ADMIN' | 'CUSTOMER' | 'TECHNICAL'>('CUSTOMER');
  const [departmentId, setDepartmentId] = useState<string>('');

  const openCreateModal = () => {
    setEditingUser(null);
    setUsername('');
    setPassword('vks@2026');
    setFullName('');
    setEmail('');
    setPhone('');
    setRole('CUSTOMER');
    setDepartmentId(departments[0]?.id || '');
    setIsModalOpen(true);
  };

  const openEditModal = (target: User) => {
    setEditingUser(target);
    setUsername(target.username);
    setPassword('');
    setFullName(target.fullName);
    setEmail(target.email || '');
    setPhone(target.phone || '');
    setRole((target.role as any) || 'CUSTOMER');
    setDepartmentId(target.departmentId || departments[0]?.id || '');
    setIsModalOpen(true);
  };

  const handleSave = async (e: React.FormEvent) => {
    e.preventDefault();

    if (!username.trim() || !fullName.trim()) {
      toast.error('Vui lòng điền đầy đủ tên đăng nhập và họ tên!');
      return;
    }

    try {
      if (editingUser) {
        const res = await apiClient.updateUser(editingUser.id, {
          fullName: fullName.trim(),
          email: email.trim() || undefined,
          phone: phone.trim() || undefined,
          role,
          departmentId: departmentId || undefined,
          password: password.trim() || undefined,
        });
        if (res.success) {
          toast.success(`Cập nhật người dùng [${username}] thành công!`);
        } else {
          toast.error(res.message || 'Lỗi cập nhật');
          return;
        }
      } else {
        if (users.some((u) => u.username.toLowerCase() === username.trim().toLowerCase())) {
          toast.error('Tên đăng nhập này đã tồn tại!');
          return;
        }

        const res = await apiClient.createUser({
          username: username.trim(),
          password: password.trim() || 'vks@2026',
          fullName: fullName.trim(),
          email: email.trim() || undefined,
          phone: phone.trim() || undefined,
          role,
          departmentId: departmentId || undefined,
        } as any);

        if (res.success) {
          toast.success(`Thêm người dùng mới [${username}] thành công!`);
        } else {
          toast.error(res.message || 'Lỗi tạo user');
          return;
        }
      }

      setIsModalOpen(false);
      onRefresh();
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    }
  };

  const handleToggleActive = async (target: User) => {
    if (target.id === currentAdmin?.id) {
      toast.error('Không thể tự khóa tài khoản của chính mình!');
      return;
    }

    try {
      const res = await apiClient.toggleUserActive(target.id);
      if (res.success) {
        toast.info(res.message || `Đã đổi trạng thái [${target.username}]`);
        onRefresh();
      } else {
        toast.error(res.message || 'Lỗi');
      }
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    }
  };

  const handleDelete = async (target: User) => {
    if (target.id === currentAdmin?.id) {
      toast.error('Không thể xóa tài khoản của chính mình!');
      return;
    }

    if (!confirm(`Bạn có chắc chắn muốn xóa tài khoản "${target.fullName}" (${target.username})?`)) return;

    try {
      const res = await apiClient.deleteUser(target.id);
      if (res.success) {
        toast.success(`Đã xóa người dùng [${target.username}]`);
        onRefresh();
      } else {
        toast.error(res.message || 'Lỗi xóa');
      }
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    }
  };

import React, { useEffect } from 'react';
import { X } from 'lucide-react';

interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  title?: string;
  subtitle?: string;
  children: React.ReactNode;
  maxWidth?: 'sm' | 'md' | 'lg' | 'xl' | '2xl' | '3xl' | '4xl';
}

export const Modal: React.FC<ModalProps> = ({
  isOpen,
  onClose,
  title,
  subtitle,
  children,
  maxWidth = 'xl',
}) => {
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    if (isOpen) {
      document.body.style.overflow = 'hidden';
      window.addEventListener('keydown', handleKeyDown);
    } else {
      document.body.style.overflow = 'unset';
    }
    return () => {
      document.body.style.overflow = 'unset';
      window.removeEventListener('keydown', handleKeyDown);
    };
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  const maxWidthClasses = {
    sm: 'max-w-sm',
    md: 'max-w-md',
    lg: 'max-w-lg',
    xl: 'max-w-xl',
    '2xl': 'max-w-2xl',
    '3xl': 'max-w-3xl',
    '4xl': 'max-w-4xl',
  }[maxWidth];

  return (
    <div className="fixed inset-0 z-50 overflow-y-auto">
      {/* Backdrop */}
      <div
        className="fixed inset-0 bg-black/60 backdrop-blur-xs transition-opacity animate-in fade-in duration-200"
        onClick={onClose}
      />

      {/* Modal Container */}
      <div className="flex min-h-full items-center justify-center p-3 sm:p-4 text-center">
        <div
          className={`w-full ${maxWidthClasses} transform overflow-hidden rounded-2xl bg-white text-left align-middle shadow-2xl transition-all border border-neutral-200/80 animate-in zoom-in-95 duration-200 relative`}
          onClick={(e) => e.stopPropagation()}
        >
          {/* Header */}
          {(title || subtitle) && (
            <div className="flex items-center justify-between border-b border-neutral-100 px-6 py-4 bg-gradient-to-r from-neutral-50 to-white">
              <div>
                {title && <h3 className="text-lg font-bold text-neutral-900 tracking-tight">{title}</h3>}
                {subtitle && <p className="text-xs text-neutral-500 mt-0.5">{subtitle}</p>}
              </div>
              <button
                type="button"
                onClick={onClose}
                className="rounded-xl p-1.5 text-neutral-400 hover:bg-neutral-100 hover:text-neutral-700 transition-colors"
                title="Đóng (Esc)"
              >
                <X className="w-5 h-5" />
              </button>
            </div>
          )}

          {/* Body */}
          <div className="p-6 max-h-[85vh] overflow-y-auto">{children}</div>
        </div>
      </div>
    </div>
  );
};

import React, { createContext, useContext, useState, useCallback } from 'react';
import { CheckCircle2, AlertCircle, Info, AlertTriangle, X } from 'lucide-react';

export type ToastType = 'success' | 'error' | 'info' | 'warning';

interface ToastItem {
  id: string;
  type: ToastType;
  title?: string;
  message: string;
}

interface ToastContextType {
  showToast: (message: string, type?: ToastType, title?: string) => void;
  success: (message: string, title?: string) => void;
  error: (message: string, title?: string) => void;
  info: (message: string, title?: string) => void;
  warning: (message: string, title?: string) => void;
}

const ToastContext = createContext<ToastContextType | undefined>(undefined);

export const ToastProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [toasts, setToasts] = useState<ToastItem[]>([]);

  const removeToast = useCallback((id: string) => {
    setToasts((prev) => prev.filter((t) => t.id !== id));
  }, []);

  const showToast = useCallback(
    (message: string, type: ToastType = 'info', title?: string) => {
      const id = `${Date.now()}-${Math.random()}`;
      setToasts((prev) => [...prev, { id, type, title, message }]);
      setTimeout(() => {
        removeToast(id);
      }, 4000);
    },
    [removeToast]
  );

  const success = useCallback((message: string, title = 'Thành công') => showToast(message, 'success', title), [showToast]);
  const error = useCallback((message: string, title = 'Lỗi') => showToast(message, 'error', title), [showToast]);
  const info = useCallback((message: string, title = 'Thông báo') => showToast(message, 'info', title), [showToast]);
  const warning = useCallback((message: string, title = 'Cảnh báo') => showToast(message, 'warning', title), [showToast]);

  return (
    <ToastContext.Provider value={{ showToast, success, error, info, warning }}>
      {children}
      <div className="fixed bottom-5 right-5 z-50 flex flex-col gap-2.5 max-w-sm w-full pointer-events-none px-4 sm:px-0">
        {toasts.map((toast) => {
          const typeConfig = {
            success: {
              icon: <CheckCircle2 className="w-5 h-5 text-emerald-600 shrink-0" />,
              border: 'border-emerald-200',
              bg: 'bg-emerald-50/95',
              titleColor: 'text-emerald-900',
            },
            error: {
              icon: <AlertCircle className="w-5 h-5 text-rose-600 shrink-0" />,
              border: 'border-rose-200',
              bg: 'bg-rose-50/95',
              titleColor: 'text-rose-900',
            },
            warning: {
              icon: <AlertTriangle className="w-5 h-5 text-amber-600 shrink-0" />,
              border: 'border-amber-200',
              bg: 'bg-amber-50/95',
              titleColor: 'text-amber-900',
            },
            info: {
              icon: <Info className="w-5 h-5 text-sky-600 shrink-0" />,
              border: 'border-sky-200',
              bg: 'bg-sky-50/95',
              titleColor: 'text-sky-900',
            },
          }[toast.type];

          return (
            <div
              key={toast.id}
              className={`pointer-events-auto flex items-start gap-3 p-4 rounded-2xl border shadow-lg backdrop-blur-sm transition-all duration-300 animate-in fade-in slide-in-from-bottom-3 ${typeConfig.bg} ${typeConfig.border}`}
            >
              <div className="mt-0.5">{typeConfig.icon}</div>
              <div className="flex-1 min-w-0">
                {toast.title && <div className={`text-sm font-semibold ${typeConfig.titleColor}`}>{toast.title}</div>}
                <div className="text-xs text-neutral-700 mt-0.5 leading-relaxed">{toast.message}</div>
              </div>
              <button
                onClick={() => removeToast(toast.id)}
                className="text-neutral-400 hover:text-neutral-600 p-1 -mr-1 rounded-lg transition-colors"
                title="Đóng"
              >
                <X className="w-4 h-4" />
              </button>
            </div>
          );
        })}
      </div>
    </ToastContext.Provider>
  );
};

export const useToast = () => {
  const context = useContext(ToastContext);
  if (!context) {
    throw new Error('useToast must be used within a ToastProvider');
  }

    return tickets.filter((t) => t.assigneeId === technicianId);

  const totalAssigned = myTickets.length;

  // Average rating
  const ratedTickets = myTickets.filter((t) => t.rating && t.rating > 0);
  const avgRating =
    ratedTickets.length > 0
      ? (
          ratedTickets.reduce((acc, t) => acc + (t.rating || 0), 0) / ratedTickets.length
        ).toFixed(1)
      : '5.0';

  const avgHours = '2.4 giờ';

        (t) => t.status === 'DONE' && t.completedAt && t.completedAt.startsWith(datePrefix)

        done: doneCount,

import React, { useState } from 'react';
import { Ticket } from '../../types';
import { Modal } from '../common/Modal';
import { StarRating } from './StarRating';
import { Send } from 'lucide-react';
import { apiClient } from '../../services/apiClient';
import { useToast } from '../common/Toast';

interface RatingModalProps {
  isOpen: boolean;
  onClose: () => void;
  ticket: Ticket | null;
  onRated?: () => void;
}

export const RatingModal: React.FC<RatingModalProps> = ({ isOpen, onClose, ticket, onRated }) => {
  const toast = useToast();

  const [stars, setStars] = useState<number>(5);
  const [speedRating, setSpeedRating] = useState<'Rất nhanh' | 'Đúng hạn' | 'Chậm'>('Rất nhanh');
  const [attitudeRating, setAttitudeRating] = useState<'Nhiệt tình' | 'Bình thường' | 'Không tốt'>('Nhiệt tình');
  const [effectivenessRating, setEffectivenessRating] = useState<'Triệt để' | 'Tạm ổn' | 'Chưa đạt'>('Triệt để');
  const [comment, setComment] = useState('');
  const [isSubmitting, setIsSubmitting] = useState(false);

  if (!ticket) return null;

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (stars < 1) {
      toast.warning('Vui lòng chọn số sao đánh giá!');
      return;
    }

    setIsSubmitting(true);

    try {
      const feedbackText = [
        `Tốc độ: ${speedRating}`,
        `Thái độ: ${attitudeRating}`,
        `Hiệu quả: ${effectivenessRating}`,
        comment.trim() ? `Nhận xét: ${comment.trim()}` : '',
      ].filter(Boolean).join(' • ');

      const res = await apiClient.evaluateTicket(ticket.id, {
        rating: stars,
        feedback: feedbackText,
      });

      if (res.success) {
        toast.success('Cảm ơn bạn đã gửi đánh giá!');
        if (onRated) onRated();
        onClose();
      } else {
        toast.error(res.message || 'Lỗi gửi đánh giá');
      }
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Modal
      isOpen={isOpen}
      onClose={onClose}
      title="Đánh giá chất lượng dịch vụ"
      subtitle={`Phiếu: ${ticket.code} — ${ticket.title}`}
      maxWidth="lg"
    >
      <form onSubmit={handleSubmit} className="space-y-6">
        <div className="bg-amber-50/60 rounded-2xl p-5 border border-amber-200 text-center">
          <label className="block text-xs font-bold text-amber-900 uppercase tracking-wider mb-2">
            Mức độ hài lòng chung
          </label>
          <div className="py-2">

import React, { useState } from 'react';
import { Star } from 'lucide-react';

interface StarRatingProps {
  value: number;
  onChange?: (val: number) => void;
  readOnly?: boolean;
  size?: 'sm' | 'md' | 'lg' | 'xl';
  showLabel?: boolean;
}

export const StarRating: React.FC<StarRatingProps> = ({
  value,
  onChange,
  readOnly = false,
  size = 'md',
  showLabel = false,
}) => {
  const [hoverValue, setHoverValue] = useState<number | null>(null);

  const starSizes = {
    sm: 'w-4 h-4',
    md: 'w-5 h-5',
    lg: 'w-7 h-7',
    xl: 'w-9 h-9',
  }[size];

  const displayVal = hoverValue !== null ? hoverValue : value;

  const getLabel = (val: number) => {
    switch (val) {
      case 5:
        return 'Rất hài lòng (5/5)';
      case 4:
        return 'Hài lòng (4/5)';
      case 3:
        return 'Bình thường (3/5)';
      case 2:
        return 'Chưa hài lòng (2/5)';
      case 1:
        return 'Rất kém (1/5)';
      default:
        return 'Chưa đánh giá';
    }
  };

  return (
    <div className="flex flex-col items-center gap-1.5">
      <div className="flex items-center gap-1">
        {[1, 2, 3, 4, 5].map((star) => {
          const filled = star <= displayVal;
          return (
            <button
              key={star}
              type="button"
              disabled={readOnly}
              onClick={() => onChange && onChange(star)}
              onMouseEnter={() => !readOnly && setHoverValue(star)}
              onMouseLeave={() => !readOnly && setHoverValue(null)}
              className={`p-0.5 transition-transform duration-150 ${
                readOnly
                  ? 'cursor-default'
                  : 'cursor-pointer hover:scale-115 focus:outline-none'
              }`}
            >
              <Star
                className={`${starSizes} ${
                  filled
                    ? 'fill-[#FFD700] text-[#FFD700] drop-shadow-xs'
                    : 'text-neutral-300 hover:text-amber-200'
                } transition-colors`}
              />
            </button>
          );
        })}
      </div>
      {showLabel && (
        <span className="text-xs font-semibold text-neutral-600 mt-1">
          {getLabel(displayVal)}
        </span>
      )}
    </div>
  );
};

interface TicketFormProps {
  onSuccess: () => void;
}

export const TicketForm: React.FC<TicketFormProps> = ({ onSuccess }) => {
  const { user } = useAuth();
  const toast = useToast();

  const [creatorName, setCreatorName] = useState(user?.fullName || '');
  const [creatorPhone, setCreatorPhone] = useState(user?.phone || '');
  const [creatorPosition, setCreatorPosition] = useState(CREATOR_POSITIONS[1]);
  const [category, setCategory] = useState(INCIDENT_CATEGORIES[0]);
  const [title, setTitle] = useState('');
  const [description, setDescription] = useState('');
  const [priority, setPriority] = useState<TicketPriority>('NORMAL');
  const [images, setImages] = useState<File[]>([]);
  const [previewUrls, setPreviewUrls] = useState<string[]>([]);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [isDragging, setIsDragging] = useState(false);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    if (!e.target.files) return;
    processFiles(Array.from(e.target.files));
  };

  const processFiles = (files: File[]) => {
    const validFiles = files.filter((f) => f.type.startsWith('image/')).slice(0, 4 - images.length);
    if (validFiles.length === 0) {
      toast.warning('Chỉ chấp nhận tệp hình ảnh (tối đa 4 ảnh)');
      return;
    }
    setImages((prev) => [...prev, ...validFiles]);
    validFiles.forEach((file) => {
      const reader = new FileReader();
      reader.onload = (event) => {
        if (event.target?.result) {
          setPreviewUrls((prev) => [...prev, event.target!.result as string]);
        }
      };
      reader.readAsDataURL(file);
    });
  };

  const handleDragOver = (e: React.DragEvent) => { e.preventDefault(); setIsDragging(true); };
  const handleDragLeave = () => setIsDragging(false);
  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    setIsDragging(false);
    if (e.dataTransfer.files) processFiles(Array.from(e.dataTransfer.files));
  };

  const removeImage = (index: number) => {
    setImages((prev) => prev.filter((_, i) => i !== index));
    setPreviewUrls((prev) => prev.filter((_, i) => i !== index));
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    if (!title.trim()) { toast.error('Vui lòng nhập tiêu đề sự cố!'); return; }
    if (!description.trim()) { toast.error('Vui lòng nhập mô tả sự cố!'); return; }

        title: title.trim(),
        description: description.trim(),
        category,
        priority,
        creatorName: creatorName.trim() || user?.fullName || 'Khách hàng',
        creatorPhone: creatorPhone.trim(),
        creatorPosition,
        supplies: [],
      });

      if (!res.success || !res.ticket) {
        toast.error(res.message || 'Lỗi gửi phiếu');
        return;
      }

      const newTicketId = res.ticket.id;

      // Upload ảnh nếu có
      for (const file of images) {
        try {
          await apiClient.uploadAttachment(newTicketId, file, 'BEFORE');
        } catch (uploadErr) {
          console.error('Lỗi upload ảnh:', uploadErr);
        }
      }

      toast.success(`Gửi phiếu [${res.ticket.code}] thành công!`);
      setTitle('');
      setDescription('');
      setImages([]);
      setPreviewUrls([]);
      onSuccess();
    } catch (err: any) {
      toast.error(err.message || 'Lỗi kết nối');
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <form onSubmit={handleSubmit} className="bg-white rounded-2xl p-6 sm:p-8 border border-neutral-200/90 shadow-sm max-w-4xl mx-auto space-y-6">
      <div className="border-b border-neutral-100 pb-4">

        </p>
      </div>

      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

            className="w-full px-3.5 py-2.5 bg-neutral-100 border border-neutral-200 rounded-xl text-neutral-600 text-sm font-medium cursor-not-allowed"
          />

            value={creatorName}

import React, { createContext, useContext, useState, useEffect } from 'react';
import { User, UserRole } from '../types';
import { apiClient } from '../services/apiClient';

interface AuthContextType {
  user: User | null;
  role: UserRole | null;
  isAuthenticated: boolean;
  login: (username: string, password?: string) => Promise<{ success: boolean; message?: string; role?: UserRole }>;
  logout: () => Promise<void>;
  refreshUser: () => Promise<void>;
}

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<User | null>(() => apiClient.getCurrentUser());

  // Load user khi app khởi động
  useEffect(() => {
    const init = async () => {
      const token = apiClient.getToken();
      if (!token) return;
      try {
        const me = await apiClient.me();
        if (me) {
          setUser(me);
        } else {
          apiClient.clearToken();
          setUser(null);
        }
      } catch {
        apiClient.clearToken();
        setUser(null);
      }
    };
    init();
  }, []);

  const login = async (username: string, password?: string) => {
    try {
      const res = await apiClient.login(username, password || '');
      if (res.success && res.user) {
        setUser(res.user);
        return { success: true, role: res.user.role };
      }
      return { success: false, message: 'Đăng nhập thất bại' };
    } catch (err: any) {
      return { success: false, message: err.message || 'Lỗi kết nối' };
    }
  };

  const logout = async () => {
    await apiClient.logout();
    setUser(null);
  };

  const refreshUser = async () => {
    const me = await apiClient.me();
    if (me) setUser(me);
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        role: user?.role || null,
        isAuthenticated: !!user,
        login,
        logout,
        refreshUser,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth must be used within AuthProvider');
  return context;
};

// ============================================
// REQUEST CORE
// ============================================
async function request<T>(url: string, options: RequestInit = {}): Promise<T> {
  const token = localStorage.getItem(TOKEN_KEY);

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...((options.headers as Record<string, string>) || {}),
  };
  if (token) headers['Authorization'] = `Bearer ${token}`;

  const res = await fetch(`${API_BASE}${url}`, { ...options, headers });

  let body: any = null;
  const ct = res.headers.get('content-type');
  try {
    body = ct?.includes('application/json') ? await res.json() : await res.text();
  } catch { body = null; }

  if (!res.ok) {
    if (res.status === 401) {
      localStorage.removeItem(TOKEN_KEY);
      localStorage.removeItem(USER_KEY);
    }
    const message =
      (typeof body === 'object' && body?.message) ||
      (typeof body === 'string' && body) ||
      `Lỗi server (${res.status})`;
    const err: any = new Error(message);
    err.status = res.status;
    throw err;
  }

  return body as T;
}

    localStorage.removeItem(USER_KEY);

      user: User;

    });
    if (data.success && data.token) {

      localStorage.setItem(USER_KEY, JSON.stringify(data.user));
    }
    return data;
  },

import { Ticket, User, Department, AuditLog, TicketStatus, RatingFeedback, SupplyItem } from '../types';
import { INITIAL_TICKETS, INITIAL_USERS, INITIAL_DEPARTMENTS, INITIAL_AUDIT_LOGS } from './mockData';

const TICKETS_KEY = 'cds_tickets_v1';
const USERS_KEY = 'cds_users_v1';
const DEPTS_KEY = 'cds_depts_v1';
const AUDIT_KEY = 'cds_audit_logs_v1';
const EVENT_NAME = 'cds_storage_sync';

export const storageService = {
  init() {
    if (!localStorage.getItem(TICKETS_KEY)) {
      localStorage.setItem(TICKETS_KEY, JSON.stringify(INITIAL_TICKETS));
    }
    if (!localStorage.getItem(USERS_KEY)) {
      localStorage.setItem(USERS_KEY, JSON.stringify(INITIAL_USERS));
    }
    if (!localStorage.getItem(DEPTS_KEY)) {
      localStorage.setItem(DEPTS_KEY, JSON.stringify(INITIAL_DEPARTMENTS));
    }
    if (!localStorage.getItem(AUDIT_KEY)) {
      localStorage.setItem(AUDIT_KEY, JSON.stringify(INITIAL_AUDIT_LOGS));
    }
  },

  notifyChange() {
    window.dispatchEvent(new CustomEvent(EVENT_NAME));
  },

  onSync(callback: () => void) {
    const handleStorage = () => callback();
    const handleCustom = () => callback();
    window.addEventListener('storage', handleStorage);
    window.addEventListener(EVENT_NAME, handleCustom);
    return () => {
      window.removeEventListener('storage', handleStorage);
      window.removeEventListener(EVENT_NAME, handleCustom);
    };
  },

  // --- TICKETS ---
  getTickets(): Ticket[] {
    try {
      const data = localStorage.getItem(TICKETS_KEY);
      return data ? JSON.parse(data) : INITIAL_TICKETS;
    } catch {
      return INITIAL_TICKETS;
    }
  },

  getTicketById(id: string): Ticket | undefined {
    return this.getTickets().find((t) => t.id === id);
  },

  saveTickets(tickets: Ticket[]) {
    localStorage.setItem(TICKETS_KEY, JSON.stringify(tickets));
    this.notifyChange();
  },

  createTicket(ticket: Omit<Ticket, 'id' | 'code' | 'createdAt'>, actorName: string): Ticket {
    const tickets = this.getTickets();
    const nextSeq = tickets.length + 1;
    const year = new Date().getFullYear();
    const code = `YCKT-${year}-${String(nextSeq).padStart(3, '0')}`;
    const newTicket: Ticket = {
      ...ticket,
      id: `t-${Date.now()}`,
      code,
      createdAt: new Date().toISOString(),
      supplies: ticket.supplies || [],
      beforeImages: ticket.beforeImages || [],
      afterImages: ticket.afterImages || [],
    };

    const updated = [newTicket, ...tickets];
    this.saveTickets(updated);

    this.addAuditLog({
      user: actorName,
      action: `Tạo phiếu yêu cầu kỹ thuật mới [${code}] - ${ticket.title}`,
      entity: `Ticket: ${code}`,
      level: 'INFO',
    });

    return newTicket;
  },

  assignTicket(ticketId: string, assigneeId: string, assigneeName: string, scheduledAt: string | undefined, actorName: string): Ticket | null {
    const tickets = this.getTickets();
    const index = tickets.findIndex((t) => t.id === ticketId);
    if (index === -1) return null;

    const ticket = tickets[index];
    const updatedTicket: Ticket = {
      ...ticket,
      assigneeId,
      assigneeName,
      scheduledAt: scheduledAt || ticket.scheduledAt,
      status: 'ASSIGNED',
      updatedAt: new Date().toISOString(),
    };

    tickets[index] = updatedTicket;
    this.saveTickets(tickets);

    this.addAuditLog({
      user: actorName,
      action: `Điều phối phiếu [${ticket.code}] cho kỹ thuật viên ${assigneeName}`,
      entity: `Ticket: ${ticket.code}`,
      level: 'INFO',
    });

    return updatedTicket;
  },

  updateTicketProcess(
    ticketId: string,
    updates: {
      status?: TicketStatus;
      scheduledAt?: string;
      technicalNotes?: string;
      supplies?: SupplyItem[];
      afterImages?: string[];
      completedAt?: string;
    },
    actorName: string
  ): Ticket | null {
    const tickets = this.getTickets();
    const index = tickets.findIndex((t) => t.id === ticketId);
    if (index === -1) return null;

    const ticket = tickets[index];
    const isCompleted = updates.status === 'DONE';

    const updatedTicket: Ticket = {
      ...ticket,
      ...updates,
      completedAt: isCompleted ? (updates.completedAt || new Date().toISOString()) : ticket.completedAt,
      updatedAt: new Date().toISOString(),
    };

    tickets[index] = updatedTicket;
    this.saveTickets(tickets);

    const statusNote = updates.status ? ` - Trạng thái: ${updates.status}` : '';
    this.addAuditLog({
      user: actorName,
      action: `Kỹ thuật viên cập nhật tiến độ xử lý phiếu [${ticket.code}]${statusNote}`,
      entity: `Ticket: ${ticket.code}`,
      level: isCompleted ? 'INFO' : 'WARNING',
    });

    return updatedTicket;
  },

  rateTicket(ticketId: string, feedback: RatingFeedback, actorName: string): Ticket | null {
    const tickets = this.getTickets();
    const index = tickets.findIndex((t) => t.id === ticketId);
    if (index === -1) return null;

    const ticket = tickets[index];
    const updatedTicket: Ticket = {
      ...ticket,
      rating: feedback.stars,
      feedback: {
        ...feedback,
        submittedAt: new Date().toISOString(),
      },
      updatedAt: new Date().toISOString(),
    };

    tickets[index] = updatedTicket;
    this.saveTickets(tickets);

    this.addAuditLog({
      user: actorName,
      action: `Khách hàng đánh giá ${feedback.stars} sao cho phiếu hoàn thành [${ticket.code}]`,
      entity: `Ticket: ${ticket.code}`,
      level: 'INFO',
    });

    return updatedTicket;
  },

  // --- USERS ---
  getUsers(): User[] {
    try {
      const data = localStorage.getItem(USERS_KEY);
      return data ? JSON.parse(data) : INITIAL_USERS;
    } catch {
      return INITIAL_USERS;
    }
  },

  saveUsers(users: User[]) {
    localStorage.setItem(USERS_KEY, JSON.stringify(users));
    this.notifyChange();
  },

  createUser(user: Omit<User, 'id'>, actorName: string): User {
    const users = this.getUsers();
    const newUser: User = {
      ...user,
      id: `user-${Date.now()}`,
    };
    const updated = [...users, newUser];
    this.saveUsers(updated);

    this.addAuditLog({
      user: actorName,
      action: `Thêm người dùng mới: ${newUser.fullName} (${newUser.username})`,
      entity: `User: ${newUser.username}`,
      level: 'SECURITY',
    });

    return newUser;
  },

  updateUser(id: string, updates: Partial<User>, actorName: string): User | null {
    const users = this.getUsers();
    const index = users.findIndex((u) => u.id === id);
    if (index === -1) return null;

    const updatedUser = { ...users[index], ...updates };
    users[index] = updatedUser;
    this.saveUsers(users);

    this.addAuditLog({
      user: actorName,
      action: `Cập nhật thông tin người dùng: ${updatedUser.fullName} (${updatedUser.username})`,
      entity: `User: ${updatedUser.username}`,
      level: 'INFO',
    });

    return updatedUser;
  },

  toggleUserActive(id: string, actorName: string): User | null {
    const users = this.getUsers();
    const index = users.findIndex((u) => u.id === id);
    if (index === -1) return null;

    const user = users[index];
    user.active = !user.active;
    users[index] = user;
    this.saveUsers(users);

    this.addAuditLog({
      user: actorName,
      action: `${user.active ? 'Mở khóa' : 'Khóa'} tài khoản người dùng: ${user.username}`,
      entity: `User: ${user.username}`,
      level: user.active ? 'INFO' : 'DANGER',
    });

    return user;
  },

  deleteUser(id: string, actorName: string): boolean {
    const users = this.getUsers();
    const target = users.find((u) => u.id === id);
    if (!target) return false;

    const updated = users.filter((u) => u.id !== id);
    this.saveUsers(updated);

    this.addAuditLog({
      user: actorName,
      action: `Xóa tài khoản người dùng: ${target.fullName} (${target.username})`,
      entity: `User: ${target.username}`,
      level: 'DANGER',
    });

    return true;
  },

  // --- DEPARTMENTS ---
  getDepartments(): Department[] {
    try {
      const data = localStorage.getItem(DEPTS_KEY);
      return data ? JSON.parse(data) : INITIAL_DEPARTMENTS;
    } catch {
      return INITIAL_DEPARTMENTS;
    }
  },

  saveDepartments(depts: Department[]) {
    localStorage.setItem(DEPTS_KEY, JSON.stringify(depts));
    this.notifyChange();
  },

  createDepartment(dept: Omit<Department, 'id'>, actorName: string): Department {
    const depts = this.getDepartments();
    const newDept: Department = {
      ...dept,
      id: `dept-${Date.now()}`,
    };
    const updated = [...depts, newDept];
    this.saveDepartments(updated);

    this.addAuditLog({
      user: actorName,
      action: `Thêm đơn vị/phòng ban mới: [${newDept.code}] ${newDept.name}`,
      entity: `Dept: ${newDept.code}`,
      level: 'INFO',
    });

    return newDept;
  },

  updateDepartment(id: string, updates: Partial<Department>, actorName: string): Department | null {
    const depts = this.getDepartments();
    const index = depts.findIndex((d) => d.id === id);
    if (index === -1) return null;

    const updatedDept = { ...depts[index], ...updates };
    depts[index] = updatedDept;
    this.saveDepartments(depts);

    this.addAuditLog({
      user: actorName,
      action: `Cập nhật thông tin phòng ban: [${updatedDept.code}] ${updatedDept.name}`,
      entity: `Dept: ${updatedDept.code}`,
      level: 'INFO',
    });

    return updatedDept;
  },

  deleteDepartment(id: string, actorName: string): boolean {
    const depts = this.getDepartments();
    const target = depts.find((d) => d.id === id);
    if (!target) return false;

    const updated = depts.filter((d) => d.id !== id);
    this.saveDepartments(updated);

    this.addAuditLog({
      user: actorName,
      action: `Xóa phòng ban [${target.code}] ${target.name}`,
      entity: `Dept: ${target.code}`,
      level: 'WARNING',
    });

    return true;
  },

  // --- AUDIT LOGS ---
  getAuditLogs(): AuditLog[] {
    try {
      const data = localStorage.getItem(AUDIT_KEY);
      return data ? JSON.parse(data) : INITIAL_AUDIT_LOGS;
    } catch {
      return INITIAL_AUDIT_LOGS;
    }
  },

  addAuditLog(entry: Omit<AuditLog, 'id' | 'timestamp' | 'ip'> & { ip?: string }) {
    const logs = this.getAuditLogs();
    const newLog: AuditLog = {
      id: `log-${Date.now()}-${Math.random().toString(36).substr(2, 4)}`,
      timestamp: new Date().toISOString(),
      ip: entry.ip || '192.168.1.100',
      ...entry,
    };
    const updated = [newLog, ...logs].slice(0, 100); // keep recent 100 logs
    localStorage.setItem(AUDIT_KEY, JSON.stringify(updated));
    this.notifyChange();
  },
};
